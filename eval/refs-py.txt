### is_prime
def is_prime(n: int) -> bool:
    if n < 2:
        return False
    i = 2
    while i * i <= n:
        if n % i == 0:
            return False
        i += 1
    return True
### gcd
def gcd(a: int, b: int) -> int:

### binary_search
def binary_search(xs: list[int], t: int) -> int:
    lo, hi = 0, len(xs)
    while lo < hi:
        mid = (lo + hi) // 2
        if xs[mid] < t:
            lo = mid + 1
        else:
            hi = mid
    return lo if lo < len(xs) and xs[lo] == t else -1

### safe_div
def safe_div(a: int, b: int) -> int:
    if b == 0:
        raise ValueError("division by zero")
    if a == -2**63 and b == -1:
        raise ValueError("overflow")
    q = abs(a) // abs(b)
    return q if (a < 0) == (b < 0) else -q

### classify
from enum import Enum

class Tri(Enum):
    EQUILATERAL = 1
    ISOSCELES = 2
    SCALENE = 3
    INVALID = 4

def classify(a: int, b: int, c: int) -> Tri:
    x, y, z = sorted((a, b, c))
    if x <= 0 or x + y <= z:
        return Tri.INVALID
    if x == z:
        return Tri.EQUILATERAL
    if x == y or y == z:
        return Tri.ISOSCELES
    return Tri.SCALENE

### bank
from dataclasses import dataclass

@dataclass
class Deposit:
    amount: int

@dataclass
class Withdraw:
    amount: int

Tx = Deposit | Withdraw

def apply(balance: int, txs: list[Tx]) -> int:
    for t in txs:
        if t.amount <= 0:
            raise ValueError("invalid amount")
        if isinstance(t, Deposit):
            balance += t.amount
        else:
            if t.amount > balance:
                raise ValueError("insufficient funds")
            balance -= t.amount
    return balance
### stack_calc
from dataclasses import dataclass

@dataclass
class Push:
    value: int

@dataclass
class Add:
    pass

@dataclass
class Mul:
    pass

@dataclass
class Neg:
    pass

Op = Push | Add | Mul | Neg

def calc(ops: list[Op]) -> int:
    st: list[int] = []
    for op in ops:
        match op:
            case Push(v):
                st.append(v)
            case Add() | Mul():
                if len(st) < 2:
                    raise ValueError("underflow")
                b, a = st.pop(), st.pop()
                st.append(a + b if isinstance(op, Add) else a * b)
            case Neg():
                if not st:
                    raise ValueError("underflow")
                st[-1] = -st[-1]
    if not st:
        raise ValueError("empty")
    return st[-1]

### median
def median(xs: list[int]) -> float:
    s = sorted(xs)
    n = len(s)
    if n % 2:
        return float(s[n // 2])
    return (s[n // 2 - 1] + s[n // 2]) / 2

def local_peaks(xs: list[int]) -> int:

        if (i == 0 or x > xs[i - 1]) and (i == len(xs) - 1 or x > xs[i + 1]):

### transfer
def transfer(bal: list[int], src: int, dst: int, amt: int) -> int:
    if not (0 <= src < len(bal) and 0 <= dst < len(bal)):
        raise ValueError("bad account")
    if amt <= 0:
        raise ValueError("bad amount")
    if bal[src] < amt:
        raise ValueError("insufficient funds")
    bal[src] -= amt
    bal[dst] += amt
    return bal[src]
### transfer_all
from dataclasses import dataclass

@dataclass
class Transfer:
    src: int
    dst: int
    amt: int

def transfer_all(bal: list[int], ts: list[Transfer]) -> int:
    work = list(bal)
    for t in ts:
        if not (0 <= t.src < len(work) and 0 <= t.dst < len(work)):
            raise ValueError("bad account")
        if t.amt <= 0:
            raise ValueError("bad amount")
        if work[t.src] < t.amt:
            raise ValueError("insufficient funds")
        work[t.src] -= t.amt
        work[t.dst] += t.amt
    bal[:] = work
    return len(ts)
### avg_i64
def avg_i64(xs: list[int]) -> int:
    if not xs:
        raise ValueError("empty")
    s = sum(xs)
    return abs(s) // len(xs) * (1 if s >= 0 else -1)
### hist
def hist(xs: list[int], lo: int, hi: int, n: int) -> list[int]:
    w = (hi - lo) // n
    out = [0] * n
    for x in xs:
        if lo <= x < hi:
            out[(x - lo) // w] += 1
    return out
